[@_CK_applystoregem]
能存放红、绿、蓝三种颜色的宝石。\
<返回/@_CK_goldsave>
#act
putitem 提供宝石 @applystoregem

[@applystoregem]
#if
US_CHECKITEMNAME 红宝石
#act
us_delete 红宝石
INC RedGem
systemmsgex $CC_SPECIAL $username  "储存红宝石成功"
#elseif
US_CHECKITEMNAME 蓝宝石
#act
us_delete 蓝宝石
INC BlueGem
systemmsgex $CC_SPECIAL $username  "储存蓝宝石成功"
#elseif
US_CHECKITEMNAME 绿宝石
#act
us_delete 绿宝石
INC BerylGem
systemmsgex $CC_SPECIAL $username  "储存绿宝石成功"
#elsesay
你提供的物品不正确！\\
<离 开/@exit>
#end
